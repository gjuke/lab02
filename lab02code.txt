int main()
{
    int x, y;
    cout << "podaj 2 inty: ";
    cin >> x >> y;
    if (x > y) {
        int dif = x - y;
        for(int i=0; i<dif; i++) {
            if (i % 3 == 0) {
                cout << i << endl;
            }
        }
    } 
    else
    if (x < y) {
        int dif = y - x;
        for (int i = 0; i < dif; i++) {
            if (i % 3 == 0) {
                cout << i << endl;
            }
        }
    }
    else {
        cout << "liczby są takie same";
        return 0;
    }
}
